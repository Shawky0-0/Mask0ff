    hits: dict[str, list[str]] = {}

        matched = role_matches(text, keywords)
        if matched:
            hits[role] = matched
    transitions = []
    for source_role in ("RAW_DATA", "STORED_DATA", "METADATA", "PATH_COMPONENT", "FILESYSTEM_OBJECT"):
        if source_role not in hits:
            continue
        for target_role in (

            "WORKFLOW_INSTRUCTION",
            "TOOL_ARGUMENT",

        ):
